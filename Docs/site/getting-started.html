Diverge SDK

Getting started

Install the SDK, configure a sandbox API key, then read the shared client.

iOS (SPM)

dependencies: [
    .package(url: "https://github.com/mohamedaldahoul/Diverge-SDK-iOS.git", from: "0.1.0")
]
// Products: DivergeSDK (required), DivergeSDKUI (optional status UI)

import DivergeSDK

try Diverge.configure(
    Configuration(apiKey: "sk_sandbox_demo", environment: .sandbox)
)
print(try Diverge.shared.apiBaseURL, Diverge.version)
